cmake_minimum_required(VERSION 3.24)
project(ShareKitWindows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    message(FATAL_ERROR "The Windows bridge currently supports x64 JVMs only; configure with -A x64.")
endif()
add_library(sharekit_windows SHARED ShareKit.cpp)
target_include_directories(sharekit_windows PRIVATE "${SHAREKIT_JNI_HEADERS}" "${SHAREKIT_JNI_HEADERS}/win32")
target_compile_definitions(sharekit_windows PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sharekit_windows PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(sharekit_windows PRIVATE windowsapp comctl32)

include(CTest)
if(BUILD_TESTING)
    add_executable(ShareKitWindowsTest ../../../jvmTest/native/windows/ShareKitWindowsTest.cpp)
    target_include_directories(ShareKitWindowsTest PRIVATE "${SHAREKIT_JNI_HEADERS}" "${SHAREKIT_JNI_HEADERS}/win32")
    target_compile_definitions(ShareKitWindowsTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(ShareKitWindowsTest PRIVATE /W4 /WX /EHsc /utf-8)
    target_link_libraries(ShareKitWindowsTest PRIVATE windowsapp comctl32)
    add_test(NAME windows-native COMMAND ShareKitWindowsTest)
    set_tests_properties(windows-native PROPERTIES TIMEOUT 30)
endif()
